Shared covariate values

The following table compares missingness and values for common columns as well as the explicitly mapped renamed variables. It excludes the iso and year keys. A value difference is counted only when both files have non-missing values and the numeric values differ.

Code 
joined_data <- inner_join(
  data |>
    select(all_of(c(key_columns, comparison_map$data_column))) |>
    rename_with(~ paste0(.x, "_data"), all_of(comparison_map$data_column)),
  data_coagent |>
    select(all_of(c(key_columns, comparison_map$coagent_column))) |>
    rename_with(
      ~ paste0(.x, "_coagent"),
      all_of(comparison_map$coagent_column)
    ),
  by = key_columns
)

value_comparison <- lapply(seq_len(nrow(comparison_map)), function(i) {
  data_column <- comparison_map$data_column[i]
  coagent_column <- comparison_map$coagent_column[i]
  data_value <- joined_data[[paste0(data_column, "_data")]]
  coagent_value <- joined_data[[paste0(coagent_column, "_coagent")]]
  same_value <- (is.na(data_value) & is.na(coagent_value)) |
    (!is.na(data_value) & !is.na(coagent_value) &
       dplyr::near(data_value, coagent_value))

  tibble(
    data_column = data_column,
    coagent_column = coagent_column,
    missing_in_data = sum(is.na(data_value)),
    missing_in_coagent = sum(is.na(coagent_value)),
    values_different = sum(!same_value)
  )
}) |> bind_rows()

mismatch_details <- lapply(seq_len(nrow(comparison_map)), function(i) {
  data_column <- comparison_map$data_column[i]
  coagent_column <- comparison_map$coagent_column[i]
  data_value <- joined_data[[paste0(data_column, "_data")]]
  coagent_value <- joined_data[[paste0(coagent_column, "_coagent")]]
  differs <- !(is.na(data_value) & is.na(coagent_value)) &
    (is.na(data_value) | is.na(coagent_value) |
       !dplyr::near(data_value, coagent_value))

  tibble(
    iso = joined_data$iso[differs],
    year = joined_data$year[differs],
    data_column = data_column,
    coagent_column = coagent_column,
    data_value = data_value[differs],
    coagent_value = coagent_value[differs]
  )
}) |> bind_rows()

if (nrow(mismatch_details) == 0) {
